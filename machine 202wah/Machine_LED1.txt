#define LED_PIN_1 11 // À changer si jamais
#define LED_PIN_2 12
//Test de LED


void setup() {

  pinMode(LED_PIN_1, OUTPUT);
  pinMode(LED_PIN_2, OUTPUT);
  Serial.begin(9600);
  Serial.setTimeout(10);

}

void loop() {
  String str;
  String Connect_YES = "Connection établie";
  String Connect_NO = "Connection échouée";
  String Problem_YES = "ERREUR";
  
  if (Serial.available() > 0) {
    str = Serial.readString();
    str.trim();
    Serial.println(str);
  }
  if (str == Connect_YES)
  {
    digitalWrite(LED_PIN_1, HIGH);

  }

    if (str == Connect_NO)
  {
    digitalWrite(LED_PIN_1, HIGH);
    delay(500);
    digitalWrite(LED_PIN_1, LOW);
    delay(500); 
    digitalWrite(LED_PIN_1, HIGH);
    delay(500);
    digitalWrite(LED_PIN_1, LOW);
  }

    if (str == Problem_YES)
  {
    digitalWrite(LED_PIN_2, HIGH);
  }

}
